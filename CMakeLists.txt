cmake_minimum_required(VERSION 3.20)
project(GenericGuiBoundary LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)

add_library(gui_boundary INTERFACE)
target_include_directories(gui_boundary INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(gui_boundary INTERFACE cxx_std_20)
target_link_libraries(gui_boundary INTERFACE Threads::Threads)

function(gui_executable target source)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE gui_boundary)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endfunction()

gui_executable(gui_example examples/demo.cpp)
if(BUILD_TESTING)
    foreach(name contract bitmap runtime layout adapter application)
        gui_executable(${name}_test tests/${name}_test.cpp)
        add_test(NAME ${name} COMMAND ${name}_test)
    endforeach()
    add_test(NAME example COMMAND gui_example)

    # Each public header must work without another header being included first.
    file(GLOB public_headers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/gui/*.hpp")
    set(header_checks)
    foreach(header IN LISTS public_headers)
        get_filename_component(name "${header}" NAME)
        set(source "${CMAKE_CURRENT_BINARY_DIR}/header_${name}.cpp")
        file(WRITE "${source}" "#include <gui/${name}>\n")
        list(APPEND header_checks "${source}")
    endforeach()
    add_library(gui_header_checks OBJECT ${header_checks})
    target_link_libraries(gui_header_checks PRIVATE gui_boundary)
    set_target_properties(gui_header_checks PROPERTIES CXX_EXTENSIONS OFF)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shared_application.cpp" "#include \"${CMAKE_CURRENT_SOURCE_DIR}/examples/application.hpp\"\n")
    add_library(gui_shared_application_check OBJECT "${CMAKE_CURRENT_BINARY_DIR}/shared_application.cpp")
    target_link_libraries(gui_shared_application_check PRIVATE gui_boundary)
    set_target_properties(gui_shared_application_check PROPERTIES CXX_EXTENSIONS OFF)
endif()
